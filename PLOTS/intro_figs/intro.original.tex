% Arguments: PNG path, center x, center y, square side in bp, view number.
% Rectangular images are center-cropped to square with their aspect ratio kept.
\newsavebox{\MIPAImageBox}
\newcommand{\MIPAImage}[5]{%
  \begin{scope}[shift={(#2,#3)}]
    \pgfmathsetmacro{\MIPAHalf}{#4/2}
    \clip (-\MIPAHalf,-\MIPAHalf) rectangle (\MIPAHalf,\MIPAHalf);
    \pgfmathsetlengthmacro{\MIPASide}{#4*1bp}
    \ifnum#5=2\relax
      \pgfmathsetlengthmacro{\MIPASide}{#4*\MIPAZoom*1bp}
    \fi
    \sbox{\MIPAImageBox}{\includegraphics{#1}}%
    \ifdim\wd\MIPAImageBox>\ht\MIPAImageBox
      \sbox{\MIPAImageBox}{\includegraphics[height=\MIPASide]{#1}}%
    \else
      \sbox{\MIPAImageBox}{\includegraphics[width=\MIPASide]{#1}}%
    \fi
    \ifnum#5=1\relax
      \node[rotate=-90,transform shape,inner sep=0pt,outer sep=0pt]
        at (0,0) {\usebox{\MIPAImageBox}};
    \else
      \node[xscale=-1,transform shape,inner sep=0pt,outer sep=0pt]
        at (0,0) {\usebox{\MIPAImageBox}};
    \fi
  \end{scope}%
}

\resizebox{\linewidth}{!}{%
\begin{tikzpicture}[x=1bp,y=1bp,inner sep=0pt,outer sep=0pt]
  \path[use as bounding box] (0,-250) rectangle (504,0);
  \fill[white] (0,-250) rectangle (504,0);

  \foreach \MIPAView/\MIPACenterX in {1/129,2/375} {
    \ifnum\MIPAView=1
      \colorlet{MIPAAccent}{MIPABlue}
      \colorlet{MIPAStrong}{MIPABlueStrong}
      \colorlet{MIPALight}{MIPABlueLight}
      \let\MIPATitle\MIPAFirstTitle
    \else
      \colorlet{MIPAAccent}{MIPAOrange}
      \colorlet{MIPAStrong}{MIPAOrangeStrong}
      \colorlet{MIPALight}{MIPAOrangeLight}
      \let\MIPATitle\MIPASecondTitle
    \fi
    \node[anchor=base,text=MIPAText,
      font=\rmfamily\bfseries\fontsize{12bp}{14bp}\selectfont]
      at (\MIPACenterX,-14) {\MIPATitle};

    \begin{scope}[shift={(\MIPACenterX,-123)}]
      \draw[MIPARim,line width=.55bp] (0,0) circle[radius=88bp];
      \foreach \MIPAFraction in {.25,.5,.75,1} {
        \pgfmathsetmacro{\MIPARadius}{34.5+39*\MIPAFraction}
        \draw[MIPAGrid,line width=.42bp,dash pattern=on 1.25bp off 1.5bp]
          (0,0) circle[radius=\MIPARadius bp];
      }

      % Spokes and annular bars, behind the images.
      \foreach \MIPAFile/\MIPAFirst/\MIPASecond
        [count=\MIPAIndex from 0] in \MIPAPrototypeData {
        \ifnum\MIPAView=1
          \let\MIPAProbability\MIPAFirst
        \else
          \let\MIPAProbability\MIPASecond
        \fi
        \pgfmathsetmacro{\MIPAAngle}{90-36*\MIPAIndex}
        \pgfmathsetmacro{\MIPAStart}{\MIPAAngle-12.5}
        \pgfmathsetmacro{\MIPAEnd}{\MIPAAngle+12.5}
        \pgfmathsetmacro{\MIPAOuter}{34.5+39*\MIPAProbability/\MIPAMaxPercent}
        \colorlet{MIPABar}{MIPALight}
        \ifdim\MIPAProbability pt<20pt\else
          \colorlet{MIPABar}{MIPAStrong}
        \fi
        \draw[MIPAGrid,line width=.48bp]
          (\MIPAAngle:34.5) -- (\MIPAAngle:88);
        \path[fill=MIPABar,draw=MIPAAccent,line width=.48bp]
          (\MIPAStart:34.5) -- (\MIPAStart:\MIPAOuter)
          arc[start angle=\MIPAStart,end angle=\MIPAEnd,radius=\MIPAOuter bp]
          -- (\MIPAEnd:34.5)
          arc[start angle=\MIPAEnd,end angle=\MIPAStart,radius=34.5bp]
          -- cycle;
      }

      \filldraw[fill=white,draw=MIPARim,line width=.6bp]
        (0,0) circle[radius=34.5bp];
      \filldraw[fill=white,draw=MIPAText,line width=.8bp]
        (-24.35,-24.35) rectangle (24.35,24.35);
      \MIPAImage{\MIPACenterImage}{0}{0}{47}{\MIPAView}

      % Prototype thumbnails and percentage labels.
      \foreach \MIPAFile/\MIPAFirst/\MIPASecond
        [count=\MIPAIndex from 0] in \MIPAPrototypeData {
        \ifnum\MIPAView=1
          \let\MIPAProbability\MIPAFirst
        \else
          \let\MIPAProbability\MIPASecond
        \fi
        \pgfmathsetmacro{\MIPAAngle}{90-36*\MIPAIndex}
        \colorlet{MIPABorder}{MIPAFrame}
        \colorlet{MIPALabel}{MIPAText}
        \def\MIPABorderWidth{.55bp}
        \let\MIPAWeight\mdseries
        \ifdim\MIPAProbability pt<20pt\else
          \colorlet{MIPABorder}{MIPAAccent}
          \colorlet{MIPALabel}{MIPAAccent}
          \def\MIPABorderWidth{1.1bp}
          \let\MIPAWeight\bfseries
        \fi
        \begin{scope}[shift={(\MIPAAngle:88)}]
          \fill[white] (-14,-14) rectangle (14,14);
          \filldraw[fill=white,draw=MIPABorder,line width=\MIPABorderWidth]
            (-12.7,-12.7) rectangle (12.7,12.7);
          \MIPAImage{\MIPAImageDirectory/\MIPAFile.png}{0}{0}{24}{\MIPAView}
          \fill[white] (-9.1,-21.8) rectangle (9.1,-13.5);

      }
